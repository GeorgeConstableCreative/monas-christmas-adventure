  function tone(type, f, t, dur, dest, vol, o) {
    o = o || {}; const os = ac.createOscillator(), g = ac.createGain(); os.type = type; os.frequency.setValueAtTime(f, t);
    if (o.det) os.detune.value = o.det; if (o.to) os.frequency.exponentialRampToValueAtTime(o.to, t + (o.toT || dur));
    let last = os; if (o.lp) { const fl = ac.createBiquadFilter(); fl.type = 'lowpass'; fl.frequency.setValueAtTime(o.lp, t); if (o.lpTo) fl.frequency.exponentialRampToValueAtTime(o.lpTo, t + dur); fl.Q.value = o.q || .8; os.connect(fl); last = fl; }
    const a = o.a || .005; g.gain.setValueAtTime(0.0001, t); g.gain.exponentialRampToValueAtTime(vol, t + a); g.gain.setValueAtTime(vol * (o.sus == null ? .7 : o.sus), t + Math.max(a, dur * .6)); g.gain.exponentialRampToValueAtTime(0.0001, t + dur + (o.r || .06));
    last.connect(g); g.connect(dest); if (o.verb) g.connect(verbIn); os.start(t); os.stop(t + dur + (o.r || .06) + .05);
  }
  function synthStep(S, st, t, d16, bus) {
    const bar = Math.floor(st / 16) % S.prog.length, s = st % 16, ch = triad(S.prog[bar]), root = ch[0], soft = S.soft, L = S.loud || 1;
    // drums: a LinnDrum-ish kit with a big gated snare
    if (S.drums === 'pop' || S.drums === 'rave' || S.drums === 'chip') {
      if (s % 4 === 0) tone('sine', 150, t, .22, bus, .85 * L, { to: 42, toT: .14, sus: .05 });
      if (s === 4 || s === 12) { noiseHit(t, bus, S.drums === 'chip' ? .12 : .32, 'bandpass', 1800, .16, 1.1); tone('triangle', 190, t, .08, bus, .12, { sus: .1 }); if (S.drums !== 'chip') noiseHit(t, verbIn, .5, 'bandpass', 2200, .25, 1); }
      if (s % 2 === 0) noiseHit(t, bus, (s % 4 === 2 ? .11 : .06) * L, 'highpass', 8000, s % 4 === 2 && S.drums === 'rave' ? .1 : .03);
      if (S.drums === 'rave' && s % 4 === 2) noiseHit(t, bus, .05, 'highpass', 9000, .02);
      if (S.drums === 'rave' && (s === 4 || s === 12)) noiseHit(t + .01, bus, .2, 'bandpass', 1300, .12, 1.4);
    } else if (S.drums === 'soft') {
      if (s === 0 || s === 10) tone('sine', 110, t, .2, bus, .45, { to: 45, toT: .12, sus: .05 });
      if (s === 4 || s === 12) noiseHit(t, verbIn, .3, 'bandpass', 2400, .2, .9);
      if (s % 2 === 0) noiseHit(t, bus, .035, 'highpass', 8500, .025);
    }
    if (S.sleigh && s % 2 === 0) noiseHit(t, bus, s % 4 ? .035 : .02, 'highpass', 7200, .06);
    // bass
    if (S.bass === 'oct' && s % 2 === 0) tone('sawtooth', MIDI(root - 12 + (s % 4 === 2 ? 12 : 0)), t, d16 * 1.7, bus, .16, { lp: 700, q: 3, sus: .5 });
    if (S.bass === 'acid' && s % 2 === 0) tone('sawtooth', MIDI(root - 12 + [0, 12, 0, 7, 0, 12, 10, 12][(s / 2) % 8]), t, d16 * 1.6, bus, .18, { lp: 1600 + 900 * Math.sin(st * .2), lpTo: 300, q: 9, sus: .3 });
    if (S.bass === 'chip' && s % 2 === 0) tone('triangle', MIDI(root + (s % 4 === 2 ? 7 : 0)), t, d16 * 1.6, bus, .18, { sus: .8 });
    if (S.bass === 'long' && s === 0) tone('triangle', MIDI(root - 12), t, d16 * 15, bus, .2, { a: .05, sus: .7 });
    // arpeggio: up through the chord and an octave
    if (S.arp) { const arp = [ch[0] + 12, ch[1] + 12, ch[2] + 12, ch[0] + 24, ch[2] + 12, ch[1] + 12]; if (!soft || s % 2 === 0) tone(S.square ? 'square' : 'sawtooth', MIDI(arp[(st / (soft ? 2 : 1)) % arp.length | 0]), t, d16 * .9, bus, S.square ? .035 : .028, { lp: S.square ? 4000 : 2600, sus: .3 }); }
    // pad
    if (S.pad && s === 0) ch.forEach(m => { tone('sawtooth', MIDI(m + 12), t, d16 * 15.5, bus, .016, { a: .35, lp: 1300, det: 9, sus: .9, r: .4 }); tone('sawtooth', MIDI(m + 12), t, d16 * 15.5, bus, .016, { a: .35, lp: 1300, det: -9, sus: .9, r: .4 }); });
